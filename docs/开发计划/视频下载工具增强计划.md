async function resolveInstagram(
  url: string, 
  proxy: string
): Promise<PublicMediaResolution> {
  // 1. 提取 shortcode
  const shortcode = extractInstagramShortcode(url)
  
  // 2. 尝试 embed 页面
  const embedUrl = `https://www.instagram.com/p/${shortcode}/embed/captioned/`
  const html = await fetchText(embedUrl, proxy)
  
  // 3. 提取视频 URL
  const videoUrl = extractFromSharedData(html) || extractFromLDJson(html)
  
  if (!videoUrl) throw new Error('无法提取 Instagram 视频链接')
  
  return {
    source: 'instagram-embed',
    videoUrl,
    referer: 'https://www.instagram.com/',
    userAgent: DEFAULT_USER_AGENT
  }
}

function extractInstagramShortcode(url: string): string {
  const patterns = [
    /instagram\.com\/p\/([A-Za-z0-9_-]+)/,
    /instagram\.com\/reel\/([A-Za-z0-9_-]+)/,
    /instagram\.com\/tv\/([A-Za-z0-9_-]+)/
  ]
  for (const pattern of patterns) {
    const match = url.match(pattern)
    if (match) return match[1]
  }
  throw new Error('无效的 Instagram URL')
}

function extractFromSharedData(html: string): string | undefined {
  // window._sharedData 提取
  const match = html.match(/window\._sharedData\s*=\s*({.+?});/)
  if (!match) return undefined
  const data = JSON.parse(match[1])
  return data?.entry_data?.PostPage?.[0]?.graphql?.shortcode_media?.video_url
}
```

async function resolveFacebook(
  url: string,
  proxy: string
): Promise<PublicMediaResolution> {
  const html = await fetchText(url, proxy)
  
  // 方法1: LD+JSON
  let videoUrl = extractFromLDJson(html)
  
  // 方法2: __TAHOE_CONFIG__
  if (!videoUrl) videoUrl = extractFromTahoeConfig(html)
  
  // 方法3: browser_native 字段
  if (!videoUrl) videoUrl = extractBrowserNativeUrl(html)
  
  if (!videoUrl) throw new Error('无法提取 Facebook 视频链接')
  
  return {
    source: 'facebook-page',
    videoUrl,
    referer: 'https://www.facebook.com/',
    userAgent: DEFAULT_USER_AGENT
  }
}

function extractBrowserNativeUrl(html: string): string | undefined {
  // 查找 browser_native_hd_url 或 browser_native_sd_url
  const hdMatch = html.match(/"browser_native_hd_url":"([^"]+)"/)
  if (hdMatch) return hdMatch[1].replace(/\\u0025/g, '%').replace(/\\\//g, '/')
  
  const sdMatch = html.match(/"browser_native_sd_url":"([^"]+)"/)
  if (sdMatch) return sdMatch[1].replace(/\\u0025/g, '%').replace(/\\\//g, '/')
  
  return undefined
}

function extractFromTahoeConfig(html: string): string | undefined {
  const match = html.match(/"__TAHOE_CONFIG__",\[\],({.+?}),\d+\]/)
  if (!match) return undefined
  const config = JSON.parse(match[1])
  return config?.video_data?.[0]?.hd_src || config?.video_data?.[0]?.sd_src
}
```

function collectTikTokUrls(video: any): Array<{url: string, score: number}> {
  const candidates: Array<{url: string, score: number}> = []
  
  // 优先级1: bitrateInfo (最高码率)
  if (video.bitrateInfo) {
    for (const item of video.bitrateInfo) {
      const bitrate = item.bitrate || item.Bitrate || 0
      const playAddr = item.playAddr || item.PlayAddr
      if (playAddr) {
        candidates.push({
          url: extractUrl(playAddr),
          score: 2_000_000 + bitrate
        })
      }
    }
  }
  
  // 优先级2: playAddr
  if (video.playAddr) {
    candidates.push({
      url: extractUrl(video.playAddr),
      score: 1_000_000
    })
  }
  
  // 优先级3: downloadAddr
  if (video.downloadAddr) {
    candidates.push({
      url: extractUrl(video.downloadAddr),
      score: 100_000
    })
  }
  
  return candidates.sort((a, b) => b.score - a.score)
}
```

  const candidates = collectTikTokUrls(video)

}

function ytdlpVideoFormat(maxHeight: number): string {
  if (maxHeight === 0) {
    // 无限制：尝试最高分辨率
    return 'bv*[vcodec^=avc1]+ba[acodec=mp4a]/bv*+ba/b/best'
  }
  
  const height = `[height<=${maxHeight}]`
  
  // 优先 AVC1(H.264) + AAC，Chromium 兼容性最好
  return [
    `bv*[vcodec^=avc1]${height}+ba[acodec=mp4a]`,  // H.264 + AAC
    `bv*[vcodec^=h264]${height}+ba[acodec=mp4a]`,  // H.264 别名
    `bv*${height}+ba`,                              // 任意视频 + 最佳音频
    `b${height}`,                                   // 合并格式
    `best${height}`                                 // 兜底
  ].join('/')
}
```

// packages/core/src/public-video.ts
const PIPED_HEALTH_CACHE = new Map<string, {healthy: boolean, checkedAt: number}>()
const HEALTH_CHECK_INTERVAL = 5 * 60 * 1000 // 5分钟

async function getHealthyPipedInstances(): Promise<string[]> {
  const now = Date.now()
  const results = await Promise.allSettled(
    PIPED_INSTANCES.map(async (instance) => {
      const cached = PIPED_HEALTH_CACHE.get(instance)
      if (cached && now - cached.checkedAt < HEALTH_CHECK_INTERVAL) {
        return cached.healthy ? instance : null
      }
      
      try {
        // 快速健康检查
        await fetchJson(`${instance}/healthcheck`, '', 3000)
        PIPED_HEALTH_CACHE.set(instance, {healthy: true, checkedAt: now})
        return instance
      } catch {
        PIPED_HEALTH_CACHE.set(instance, {healthy: false, checkedAt: now})
        return null
      }
    })
  )
  
  return results
    .filter((r): r is PromiseFulfilledResult<string> => 
      r.status === 'fulfilled' && r.value !== null
    )
    .map(r => r.value)
}

```
1. H.264 (AVC1) + AAC - Chromium 最佳兼容
2. H.265 (HEVC) - 高码率，需检查播放支持
3. VP9 + Opus - WebM 容器，部分场景
4. AV1 - 最新编码，兼容性待确认
```

#### 音频要求
- **不重新编码**: 使用 ffmpeg `-c:a copy`
- **保留元数据**: `-map_metadata 0`
- **比特率**: 保持原始（通常 128-320kbps AAC）

#### FFmpeg 命令模板
```bash
# 合并视频音频（不重编码）
ffmpeg -i video.mp4 -i audio.m4a \
  -c:v copy -c:a copy \
  -map_metadata 0 \
  -movflags +faststart \
  output.mp4

# Chromium 兼容性转码（仅在必要时）
ffmpeg -i input.mp4 \
  -c:v libx264 -preset fast -crf 18 \
  -c:a aac -b:a 192k \
  -pix_fmt yuv420p \
  output.mp4
```

// packages/shared/src/index.ts
export function toUserTaskMessage(error: string): string {
  // yt-dlp 错误映射
  if (error.includes('Sign in to confirm your age')) {
    return '视频有年龄限制，需要登录。请在设置中配置 Cookie。'
  }
  if (error.includes('This video is not available')) {
    return '视频不可用（可能已删除或区域限制）'
  }
  if (error.includes('ERROR: unable to download video data')) {
    return '下载失败，请检查网络或尝试配置代理'
  }
  
  // 公开解析错误
  if (error.includes('Piped')) {
    return 'YouTube 公开解析失败，正在尝试备用方案...'
  }
  
  return error
}
