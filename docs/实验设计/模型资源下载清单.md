print("开始预下载 WhisperX 对齐模型...")

for lang in models_to_download:

        torch.cuda.empty_cache()

import torch

class ModelBenchmark:
    """模型对比实验框架"""
    
    def __init__(self, test_audio_dir: str, output_dir: str):
        self.test_audio_dir = Path(test_audio_dir)
        self.output_dir = Path(output_dir)
        self.output_dir.mkdir(exist_ok=True)
        
        self.results = []
    
    def benchmark_all(self):
        """运行所有模型对比"""
        
        test_cases = [
            ("japanese/clean_speech_5min.wav", "japanese/clean_speech_5min.txt", "ja"),
            ("chinese/mandarin_standard_5min.wav", "chinese/mandarin_standard_5min.txt", "zh"),
            ("taiwanese/minnan_pure_5min.wav", "taiwanese/minnan_pure_5min.txt", "nan"),
        ]
        
        for audio_file, script_file, language in test_cases:
            audio_path = self.test_audio_dir / audio_file
            script_path = self.test_audio_dir / script_file
            
            if not audio_path.exists():
                print(f"⚠️ 跳过: {audio_file} (文件不存在)")
                continue
            
            script_text = script_path.read_text(encoding='utf-8') if script_path.exists() else None
            
            print(f"\n{'=' * 60}")
            print(f"测试文件: {audio_file}")
            print(f"语言: {language}")
            print(f"有原文: {'是' if script_text else '否'}")
            print(f"{'=' * 60}\n")
            
            # 1. Faster-Whisper
            self.run_faster_whisper(audio_path, language, script_text)
            
            # 2. stable-ts
            self.run_stable_ts(audio_path, language, script_text)
            
            # 3. WhisperX
            if language in ["ja", "zh"]:
                self.run_whisperx(audio_path, language, script_text)
